# ========== VERIFY JOURNAL BALANCED ==========
JournalEntry::with('details')->get()->map(function($e) {
    return [
        'entry' => $e->entry_number,
        'debit' => $e->details->sum('debit'),
        'credit' => $e->details->sum('credit'),
        'balanced' => $e->details->sum('debit') == $e->details->sum('credit')
    ];
});
